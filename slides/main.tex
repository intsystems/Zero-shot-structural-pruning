\begin{frame}{Структурное прореживание моделей глубокого обучения}
    \begin{block}{Цель} 
      Предложить метод структурного прореживания нейросетей.
    \end{block}
    
    ~\\
    \begin{block}{Проблема}
      Существующие методы работают послойно. Учёт взаимосвязей между слоями модели является вычислительно сложной задачей.
    \end{block}
    ~\\
    \begin{block}{Метод решения}
      Предлагаемый метод основан на построении суррогатной графовой модели, повторяющей структуру исходной нейросети.
    \end{block}
\end{frame}

\begin{frame}{Постановка задачи}
    \begin{itemize}
        \item Архитектура модели представляет собой ориентированный ациклический граф $G = (V, E)$. Каждая вершина $v_i \in V$ соответствует слою нейронной сети, и каждое ребро $e_{ij} \in E$ представляет поток данных из выхода слоя $v_i$ до входа $v_j$. 
        
        \item В данной формулировке удаление вершины означает удаление соответствующего слоя из сети, тогда как удаление ребра к устранению зависимости передачи данных между слоями (например, в случае residual connection).
        
        \item Задана выборка $\mathfrak{D} = \mathfrak{D}_\text{train} \cup \mathfrak{D}_\text{val}$. Задана функция потерь $\mathcal{L}$. Пусть $\boldsymbol w$ -- параметры модели до прунинга, а $\boldsymbol w'$ -- параметры модели после. Задача оптимизации
        $$\min_{\boldsymbol w' \in \mathbb{R}^k} \; \left| \mathcal{L}(\mathfrak{D} \mid \boldsymbol w') - \mathcal{L}(\mathfrak{D} \mid \boldsymbol w) \right|.$$
    
    \end{itemize}
\end{frame}

\begin{frame}{Предлагаемый метод}
    \begin{itemize}
        \item Пусть задано $\boldsymbol{\gamma}_{ij} \in [0, 1]$ для каждого ребра $e_{ij} \in E$. Также для графа $G$ задана маска рёбер $\mathbf{M} \in \mathbb{Z}_2^{|E|}$:
        $$ \mathbf{M}_{ij} = 0 \quad \Leftrightarrow \quad \boldsymbol{\gamma}_{ij} = 0.$$

        \item Дополнительно накладываем следующее ограничение на исходящие ребра каждой вершины:
        $$ \sum_{j:\ e_{ij} \in E} \boldsymbol{\gamma}_{ij} = 1 $$

        \item Тогда оптимизационная задача:
        $$
        \begin{aligned}
        \boldsymbol{\gamma}
        =
        \arg\min_{\boldsymbol{\gamma} \in [0,1]^{|E|}}
        \quad
        & \mathsf{E}_{\mathbf{M}}
        \left\|
        G(\boldsymbol{\gamma}, \mathbf{M})
        -
        S(\mathbf{w} \mid \mathfrak{D}, f)
        \right\|_2^2
        \\
        \textrm{s.t.} \quad
        &
        \sum_{j:\ e_{ij} \in E} \boldsymbol{\gamma}_{ij} = 1.
        \end{aligned}
        $$
    \end{itemize}
\end{frame}

\begin{frame}{Предлагаемый метод}
    \begin{theorem}{Теорема}
    Пусть задано значение $\mathbf{M}=\mathbf{1}$, соответствующее модели со всеми рёбрами, то есть
    \[
    \forall (i,j)\in E:\ \mathbf{M}_{ij}=1.
    \]
    
    Тогда в окрестности $\mathbf{M}=\mathbf{1}$ функция ошибки $S$ аппроксимируется формулой, соответствующей графу:
    
    \[
    \begin{split}
    S(\mathbf{M}) =
    &\sum_{(i,j) \in E}
    \frac{\partial G}{\partial \mathbf{M}_{ij}}
    \Bigg|_{\mathbf{M}=\mathbf{1}}
    (\mathbf{M}_{ij} - 1)
    \\
    &+ \frac{1}{2}
    \sum_{(i,j) \in E}
    \sum_{(k,l) \in E}
    \frac{\partial^2 G}
    {\partial \mathbf{M}_{ij}\,\partial \mathbf{M}_{kl}}
    \Bigg|_{\mathbf{M}=\mathbf{1}}
    (\mathbf{M}_{ij} - 1)(\mathbf{M}_{kl} - 1)
    \\
    &+ o(\|\mathbf{M} - \mathbf{1}\|^2).
    \end{split}
    \]
    \end{theorem}
\end{frame}

\begin{frame}{Вычислительный эксперимент}
    \begin{columns}
        \begin{column}{0.5\textwidth}
            \begin{itemize}
            \item Рассматриваются задачи классификации изображений из датасета MNIST.

            \vspace{10pt}
            
            \item В сравнении участвуют следующие базовые методы:
                \begin{small}
                \begin{itemize} 
                \item \textbf{(Graph)}: Предложенный метод,
                \item \textbf{(FCN)}: Простая нейросеть,
                \item \textbf{(Linear)}: Линейная регрессия,
                \item \textbf{(L1)}: Метод, основанный на измерении величин параметров,
                \item \textbf{(Taylor)}: Метод, использующий приближение Тейлора до 1-го и 2-го порядков.
                \end{itemize}
              \end{small}
          \end{itemize}
        \end{column}
